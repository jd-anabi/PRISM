class Theme:
    """The one switch. TH["ink"] is the hex string, TH.q("ink") the QColor, for the theme in force."""
    dark = False

    def __getitem__(self, name):
        return COLOURS[name][1 if self.dark else 0]

    def q(self, name):
        return QColor(self[name])

    def all(self):
        return {k: self[k] for k in COLOURS}

# an experiment for the report only: --hinting=none draws letters unfitted to the pixel grid, as a browser does
HINTING = next((a.split("=", 1)[1] for a in sys.argv if a.startswith("--hinting=")), "default")
_FONTS = {}


def _hint(f):
    if HINTING != "default":
        f.setHintingPreference({"none": QFont.PreferNoHinting, "vertical": QFont.PreferVerticalHinting,
                                "full": QFont.PreferFullHinting}[HINTING])
    return f


def inter(kind="body", weight=None, px=None):
    """The program's face, with figures of equal width switched on (an OpenType feature, set in code).
    Built once for each size and weight; a caller gets its own copy."""
    key = (kind, weight, px)
    if key not in _FONTS:
        size, _, w = TYPE[kind]
        f = QFont(LETTERING)
        f.setPixelSize(px or size)
        f.setWeight(QFont.Weight(weight or w))
        try:                                        # the stub says a str is accepted; in 6.9.3 only a Tag is
            f.setFeature(QFont.Tag.fromString("tnum"), 1)
        except Exception:                           # noqa: BLE001 - an older Qt: proportional figures
            pass
        _FONTS[key] = _hint(f)
    return QFont(_FONTS[key])


def book(bold=False):
    """The book face. Built in code: a style sheet can name a family but cannot switch a feature off."""
    key = ("book", bold)
    if key not in _FONTS:
        f = QFont(BOOK)
        f.setPixelSize(TYPE["book"][0])
        f.setBold(bold)
        _FONTS[key] = _hint(f)
    return QFont(_FONTS[key])

# ── palette: what a style sheet does badly (placeholder and link colours, the selection, the disabled
#    group, the colours custom-painted and un-styled parts fall back to) ──
def build_palette():
    p = QPalette()
    pairs = {
        QPalette.Window: "paper", QPalette.WindowText: "ink", QPalette.Base: "field", QPalette.AlternateBase: "margin",
        QPalette.Text: "ink", QPalette.Button: "paper", QPalette.ButtonText: "ink", QPalette.BrightText: "on_accent",
        QPalette.ToolTipBase: "raised", QPalette.ToolTipText: "ink", QPalette.PlaceholderText: "ink_3",
        QPalette.Highlight: "accent", QPalette.HighlightedText: "on_accent", QPalette.Link: "accent",
        QPalette.LinkVisited: "accent", QPalette.Mid: "rule_light", QPalette.Midlight: "margin",
        QPalette.Light: "paper", QPalette.Dark: "field_line", QPalette.Shadow: "ink",
    }
    if hasattr(QPalette, "Accent"):
        pairs[QPalette.Accent] = "accent"
    for role, name in pairs.items():
        for group in (QPalette.Active, QPalette.Inactive):
            p.setColor(group, role, TH.q(name))
        p.setColor(QPalette.Disabled, role, TH.q(name))
    for role in (QPalette.WindowText, QPalette.Text, QPalette.ButtonText, QPalette.PlaceholderText):
        p.setColor(QPalette.Disabled, role, TH.q("disabled_ink"))
    for role in (QPalette.Base, QPalette.Button):
        p.setColor(QPalette.Disabled, role, TH.q("disabled_fill"))
    p.setColor(QPalette.Disabled, QPalette.Highlight, TH.q("disabled_line"))
    return p


def set_class(widget, **props):
    """Name a widget's class (or any other property a rule of the style sheet tests) and make the sheet look
    again. A sheet matches its rules when a widget is first polished; a property set after that changes nothing
    by itself, so every class is set through here."""
    for name, value in props.items():
        widget.setProperty(name, value)
    if widget.testAttribute(Qt.WA_WState_Polished):
        style = widget.style()
        style.unpolish(widget)
        style.polish(widget)
        widget.update()
    return widget

def apply_theme(dark):
    """Palette, style sheet, the title bars Windows draws, the figure settings, then every piece that keeps
    colours of its own."""
    app = QApplication.instance()
    TH.dark = bool(dark)
    app.styleHints().setColorScheme(Qt.ColorScheme.Dark if dark else Qt.ColorScheme.Light)   # native frames follow
    app.setPalette(build_palette())
    app.setStyleSheet(build_qss())
    set_class_fonts()
    matplotlib.rcParams.update(journal_rc())
    for top in app.topLevelWidgets():
        for w in top.findChildren(QWidget):
            if hasattr(w, "retheme"):
                w.retheme()
        top.update()
    if FOLLOWER is not None:
        FOLLOWER.refresh()                           # a change of style hides the toolkit's focus frame

def paint_marker(p, kind, box, disabled=False):
    """All the markers and the few glyphs, drawn in a 16-unit box by ONE function. One shape per state."""
    q = (lambda n: TH.q("disabled_ink")) if disabled else TH.q
    p.save()
    p.setRenderHint(QPainter.Antialiasing, True)
    p.translate(box.topLeft())
    p.scale(box.width() / 16.0, box.height() / 16.0)

    def pen(name, w=2.0, cap=Qt.RoundCap):
        pn = QPen(q(name), w)
        pn.setCapStyle(cap)
        pn.setJoinStyle(Qt.RoundJoin)
        return pn

    def line(*pts):
        path = QPainterPath(QPointF(*pts[0]))
        for pt in pts[1:]:
            path.lineTo(QPointF(*pt))
        return path

    def stroke(path, name, w=2.0, cap=Qt.RoundCap):
        p.setPen(pen(name, w, cap)); p.setBrush(Qt.NoBrush); p.drawPath(path)

    def disc(cx, cy, r, fill=None, edge=None, w=2.0):
        p.setPen(pen(edge, w) if edge else Qt.NoPen)
        p.setBrush(q(fill) if fill else Qt.NoBrush)
        p.drawEllipse(QPointF(cx, cy), r, r)

    if kind == "done":                          # an ink disc with a tick
        disc(8, 8, 8, "ink")
        stroke(line((4.5, 8.4), (6.9, 10.8), (11.6, 5.6)), "paper")
    elif kind == "now":                         # a blue dot
        disc(8, 8, 6, "accent")
    elif kind == "later":                       # an open, broken circle
        pn = pen("ink_3", 2.0, Qt.FlatCap)
        pn.setDashPattern([1.04, 1.0])          # in pen widths
        p.setPen(pn); p.setBrush(Qt.NoBrush); p.drawEllipse(QPointF(8, 8), 6.5, 6.5)
    elif kind == "half":                        # unfinished: a half disc
        disc(8, 8, 7, "paper", "ink")
        half = QPainterPath(QPointF(8, 1.5))
        half.arcTo(QRectF(1.5, 1.5, 13, 13), 90, 180)
        half.closeSubpath()
        p.setPen(Qt.NoPen); p.setBrush(q("ink")); p.drawPath(half)
    elif kind == "none":                        # nothing running: a dash
        stroke(line((4, 8), (12, 8)), "ink_3")

        p.setPen(pen("warn_edge", 1.5)); p.setBrush(q("warn")); p.drawPath(tri)

    elif kind == "err":                         # a square with a cross
        p.setPen(Qt.NoPen); p.setBrush(q("err")); p.drawRoundedRect(QRectF(0, 0, 16, 16), 2, 2)

    elif kind == "lock":                        # blocked: a padlock
        p.setPen(Qt.NoPen); p.setBrush(q("ink_2")); p.drawRoundedRect(QRectF(3, 7, 10, 8), 1.5, 1.5)
        bow = QPainterPath(QPointF(5, 7.5))

        bow.arcTo(QRectF(5, 2, 6, 6), 180, -180)
        bow.lineTo(11, 7.5)

    # glyphs for controls
    elif kind == "chev_right":
        stroke(line((6, 3.5), (10.5, 8), (6, 12.5)), "ink")
    elif kind == "chev_down":
        stroke(line((3.5, 6), (8, 10.5), (12.5, 6)), "ink")
    elif kind == "help":
        disc(8, 8, 7, None, "ink")

        p.setPen(Qt.NoPen); p.setBrush(q("err")); p.drawRect(QRectF(3, 3, 10, 10))
    p.restore()


def paint_indicator(p, r, radio, on, enabled):
    """The tick box and the radio dot, in an 18-unit box, from the table. Drawn by the style (see PrismStyle)."""
    p.save()
    p.setRenderHint(QPainter.Antialiasing, True)
    k = r.width() / 18.0
    line = TH.q("field_line" if enabled else "disabled_line")
    ground = TH.q("field" if enabled else "disabled_fill")
    mark = TH.q("on_accent" if enabled else "disabled_ink")
    filled = on and enabled
    p.setPen(Qt.NoPen if filled else QPen(line, 1.0))
    p.setBrush(TH.q("accent") if filled else ground)
    body = r if filled else r.adjusted(0.5, 0.5, -0.5, -0.5)
    if radio:
        p.drawEllipse(body)
        if on:
            p.setPen(Qt.NoPen); p.setBrush(mark); p.drawEllipse(r.center(), 4 * k, 4 * k)
    else:
        corner = (3.0 if filled else 2.5) * k
        p.drawRoundedRect(body, corner, corner)
        if on:
            tick = QPainterPath(r.topLeft() + QPointF(4.975 * k, 9.35 * k))
            tick.lineTo(r.topLeft() + QPointF(7.6 * k, 11.975 * k))
            tick.lineTo(r.topLeft() + QPointF(13.025 * k, 6.025 * k))
            p.setPen(QPen(mark, 1.75 * k, Qt.SolidLine, Qt.RoundCap, Qt.RoundJoin)); p.setBrush(Qt.NoBrush)
            p.drawPath(tick)
    p.restore()


class PrismStyle(QProxyStyle):
    """Fusion, with the few things a style sheet cannot do taken from the table: the tick box and the radio
    dot are painted (no image files), the toolkit's own dotted focus mark is left out (the ring is the one
    sign of focus), and the focus frame is told how far outside its control it sits and to lie above it."""
    INDICATORS = (QStyle.PM_IndicatorWidth, QStyle.PM_IndicatorHeight,
                  QStyle.PM_ExclusiveIndicatorWidth, QStyle.PM_ExclusiveIndicatorHeight)

    def __init__(self):
        super().__init__("Fusion")

    def pixelMetric(self, metric, option=None, widget=None):
        if metric in self.INDICATORS:
            return SIZE["indicator"]
        if metric in (QStyle.PM_FocusFrameHMargin, QStyle.PM_FocusFrameVMargin):
            return 0 if getattr(widget, "inset", False) else ring_room()
        return super().pixelMetric(metric, option, widget)

    def styleHint(self, hint, option=None, widget=None, data=None):
        if hint == QStyle.SH_FocusFrame_AboveWidget:
            return 1                                 # the frame is a child of the window (or of a scroll area's
        return super().styleHint(hint, option, widget, data)   # viewport) and follows every ancestor of its control

    def drawPrimitive(self, element, option, p, widget=None):
        if element in (QStyle.PE_IndicatorCheckBox, QStyle.PE_IndicatorRadioButton):
            paint_indicator(p, QRectF(option.rect), element == QStyle.PE_IndicatorRadioButton,
                            bool(option.state & QStyle.State_On), bool(option.state & QStyle.State_Enabled))
            return
        if element == QStyle.PE_FrameFocusRect:
            return
        super().drawPrimitive(element, option, p, widget)


class PaintedIcon(QIconEngine):
    """An icon that is the marker function: sharp at any display scaling, recoloured by the theme, no files."""

    def __init__(self, kind):
        super().__init__()
        self.kind = kind

    def paint(self, painter, rect, mode, state):
        side = min(rect.width(), rect.height())       # a wider icon keeps the glyph at its left: room before the words
        box = QRectF(rect.left(), rect.top() + (rect.height() - side) / 2, side, side)
        paint_marker(painter, self.kind, box, disabled=(mode == QIcon.Disabled))

    def pixmap(self, size, mode, state):
        pm = QPixmap(size)
        pm.fill(Qt.transparent)
        painter = QPainter(pm)
        self.paint(painter, QRect(QPoint(0, 0), size), mode, state)
        painter.end()
        return pm

    def clone(self):
        return PaintedIcon(self.kind)

def fill_px(p, r, colour, from_bottom=False, from_right=False):
    """Fill a thin rectangle on whole device pixels. Its thickness is a whole number of pixels worked out ONCE,
    rounded down as a browser rounds a border (never under one pixel), and it is laid from the edge named. So a
    rule has one weight wherever it falls, it stays inside the room its widget keeps for it, and a child widget
    beside it never covers part of it. Left to the painter, a 1 px line at 150 % is 1 or 2 pixels by position."""
    t = p.deviceTransform()
    inv, _ = t.inverted()
    a, b = t.map(r.topLeft()), t.map(r.bottomRight())
    edge = lambda v: math.floor(v + 0.5)
    whole = lambda v: max(1, math.floor(v + 1e-6))
    x0, x1, y0, y1 = edge(a.x()), edge(b.x()), edge(a.y()), edge(b.y())
    if r.height() <= r.width():                       # a horizontal rule: its thickness is what must not vary
        h = whole(b.y() - a.y())
        y0, y1 = (y1 - h, y1) if from_bottom else (y0, y0 + h)
    else:
        w = whole(b.x() - a.x())
        x0, x1 = (x1 - w, x1) if from_right else (x0, x0 + w)
    p.fillRect(QRectF(inv.map(QPointF(x0, y0)), inv.map(QPointF(x1, y1))), colour)


class Ruled(QWidget):
    """A block with a ruled line along its top edge, its bottom edge or both; what it holds goes in .box.
    The block paints its own rules, from its own edges and on whole pixels, so a rule is the same weight at
    every display scaling and every position (a rule that is a widget 1 or 2 px high is not)."""

    def __init__(self, top=0, bottom=0, colour="rule", gap_top=0, gap_bottom=0):
        super().__init__()
        self._top, self._bottom, self._colour = top, bottom, colour
        self.box = QVBoxLayout(self)
        self.box.setContentsMargins(0, top + gap_top, 0, bottom + gap_bottom)
        self.box.setSpacing(0)

    def paintEvent(self, _):
        p = QPainter(self)
        w, h, colour = self.width(), self.height(), TH.q(self._colour)
        if self._top:
            fill_px(p, QRectF(0, 0, w, self._top), colour)
        if self._bottom:
            fill_px(p, QRectF(0, h - self._bottom, w, self._bottom), colour, from_bottom=True)


def paint_hatch(p, rect, colour="hatch"):
    """Diagonal 1 px strokes every 5 px across the stroke: the range in which a run should end."""
    if rect.width() <= 0:
        return
    p.save()
    p.setClipRect(rect)
    p.setRenderHint(QPainter.Antialiasing, True)
    p.setPen(QPen(TH.q(colour), 1.0))
    step, h = SIZE["hatch_step"] * math.sqrt(2.0), rect.height()
    x = rect.left() - h - (rect.left() % step)        # the strokes stay where they are as the rectangle moves
    while x < rect.right() + h:
        p.drawLine(QPointF(x, rect.bottom()), QPointF(x + h, rect.top()))
        x += step
    p.restore()

# ── the time axis: a pure layout function, then a widget that only paints what it returns ──
def axis_layout(start, now, end_from, end_by, width, text_width=None):
    """Where everything on the time axis goes, as plain numbers: no painter and no widget, so it can be tested
    without a screen. The axis runs from the start to the end of the expected range. Ticks: every 30 seconds
    under 5 minutes, every 15 minutes under 2 hours, every 6 hours on the clock under 2 days, and beyond that at
    each midnight with the name of the day between. A run that has passed its range gets an axis a day longer
    (as many days as it needs) and is reported late. The "now" mark never leaves the axis."""
    measure = text_width or (lambda s: 7.0 * len(s))
    rnd = lambda v: int(math.floor(v + 0.5))
    day = dt.timedelta(days=1)
    late = now > end_by
    axis_end = end_by + day * max(1, math.ceil((now - end_by) / day)) if late else end_by
    span = (axis_end - start).total_seconds()
    at = lambda t: (t - start).total_seconds() / span * width
    half = AXIS["mark_w"] / 2 + AXIS["mark_halo"]
    x_done = min(max(rnd(at(now)), 0), width)
    out = {"late": late, "inside": end_from <= now <= end_by, "axis_end": axis_end, "x_done": x_done,
           "x_now": min(max(x_done, half), width - half), "x_from": rnd(at(end_from)), "x_by": rnd(at(end_by))}
    step, clock = next((s, c) for limit, s, c in ((300, 30, "%H:%M:%S"), (7200, 900, "%H:%M"),
                                                  (172800, 21600, "%H:%M"), (math.inf, 86400, None)) if span < limit)
    midnight = dt.datetime.combine(start.date(), dt.time())
    k = math.floor((start - midnight).total_seconds() / step) + 1          # the first tick after the start
    times = []
    while midnight + dt.timedelta(seconds=k * step) < axis_end:
        times.append(midnight + dt.timedelta(seconds=k * step))
        k += 1
    out["step"], out["ticks"] = step, [rnd(at(t)) for t in times]
    labels, right = [], -1e9

    def put(text, x):                                  # a label is left out sooner than printed over its neighbour
        nonlocal right
        w = measure(text)
        x = min(max(x(w), 0.0), width - w)
        if x >= right + AXIS["label_gap"] or not labels:
            labels.append((x, text))
            right = x + w

    if clock:                                          # the clock under each tick; the day's name at a midnight
        for t in times:
            put(t.strftime(("%a " if t.time() == dt.time() else "") + clock), lambda w, t=t: at(t) - w / 2)
    else:                                              # the day's name between two midnights
        edges = [start] + times + [axis_end]
        for i, (a, b) in enumerate(zip(edges, edges[1:])):
            if i == 0 and a.time() != dt.time():       # the part-day the run began in: its name at the origin
                put(a.strftime("%a"), lambda w: 0.0)
            else:
                put(a.strftime("%a"), lambda w, a=a, b=b: (at(a) + at(b)) / 2 - w / 2)
    out["labels"] = labels
    return out


class TimeAxis(QWidget):
    """The one progress indication: the run drawn on a labelled time axis. Only the "now" mark moves."""

    def __init__(self, start, now, end_from, end_by):
        super().__init__()
        self.times = (start, now, end_from, end_by)
        self.setFixedHeight(AXIS["h"])

    def layout_now(self):
        return axis_layout(*self.times, self.width(), QFontMetricsF(inter("small")).horizontalAdvance)

    def paintEvent(self, _):
        p = QPainter(self)
        L, A, w = self.layout_now(), AXIS, self.width()
        top, h, hair = A["track_y"], A["track_h"], SIZE["hair"]
        rule, paper = TH.q("rule"), TH.q("paper")
        p.fillRect(QRectF(0, top, w, h), paper)
        if not L["late"]:                                                  # the range still ahead, hatched
            paint_hatch(p, QRectF(L["x_from"], top + hair, w - hair - L["x_from"], h - 2 * hair))
            fill_px(p, QRectF(L["x_from"], top, hair, h), rule)
        p.fillRect(QRectF(0, top, L["x_done"], h), TH.q("elapsed"))         # elapsed, in ink
        if L["late"] or L["inside"]:                                       # the range shows through the bar
            upto = min(L["x_done"], L["x_by"])
            paint_hatch(p, QRectF(L["x_from"], top + hair, upto - L["x_from"], h - 2 * hair), "paper")
            fill_px(p, QRectF(L["x_from"], top, hair, h), paper)
            if L["late"]:
                fill_px(p, QRectF(L["x_by"] - hair, top, hair, h), paper)
        fill_px(p, QRectF(0, top, w, hair), rule)
        fill_px(p, QRectF(0, top + h - hair, w, hair), rule, from_bottom=True)
        fill_px(p, QRectF(0, top, hair, h), rule)
        fill_px(p, QRectF(w - hair, top, hair, h), rule, from_right=True)
        x, mw, halo = L["x_now"], A["mark_w"], A["mark_halo"]               # the "now" mark
        p.fillRect(QRectF(x - mw / 2 - halo, 0, mw + 2 * halo, A["mark_h"]), paper)
        p.fillRect(QRectF(x - mw / 2, 0, mw, A["mark_h"]), TH.q("accent"))
        for tx in L["ticks"]:
            fill_px(p, QRectF(tx, A["tick_y"], hair, A["tick_h"]), rule)
        p.setFont(inter("small"))
        p.setPen(TH.q("ink_2"))
        for lx, text in L["labels"]:
            p.drawText(QRectF(lx, A["label_y"], w, A["label_h"]), Qt.AlignLeft | Qt.AlignVCenter, text)


class Progress(QWidget):
    """The axis and the one line of words under it. The words follow the same layout: the expected range,
    then "by" once the run is inside it, then the attention marker when the run is late."""

    def __init__(self, start, now, end_from, end_by):
        super().__init__()
        v = QVBoxLayout(self)
        v.setContentsMargins(0, 0, 0, 0); v.setSpacing(0)
        self.axis = TimeAxis(start, now, end_from, end_by)
        v.addWidget(self.axis)
        ends = QHBoxLayout()
        ends.setSpacing(GAP["axis_key"])
        self.started = label("")
        ends.addWidget(self.started)
        ends.addStretch(1)
        clock = "%a %H:%M" if end_by - start >= dt.timedelta(minutes=5) else "%a %H:%M:%S"
        late, inside = now > end_by, end_from <= now <= end_by
        ends.addWidget(Marker("warn") if late else HatchKey(), 0, Qt.AlignVCenter)
        self.words = label(f"Later than expected: it was due by {end_by.strftime(clock)}" if late
                           else f"Expected to end by {end_by.strftime(clock)}" if inside
                           else f"Expected to end between {end_from.strftime(clock)} and {end_by.strftime(clock)}")
        ends.addWidget(self.words)
        self.started.setText("Started " + start.strftime(clock))
        v.addLayout(ends)
        hours = lambda t: round((t - start).total_seconds() / 3600)
        self.axis.setAccessibleName(f"{hours(now)} hours of an expected {hours(end_from)} to {hours(end_by)} have passed"
                                    + (", later than expected" if late else ""))

                       Qt.AlignLeft | Qt.AlignVCenter, self.words)


# ── keyboard focus: one ring, carried by the toolkit's own focus frame ──
class FocusRing(QFocusFrame):
    """The one ring: 2 px, 2 px clear of its control. The toolkit's focus frame keeps it on the control through
    every move, resize, scroll and re-layout of any ancestor, and inside a scroll area it is a child of the
    viewport, so it scrolls with its control and is cut where the control is. A control that fills its row
    (a part of the program, a step, a scrolling page) has the ring just inside its own edge instead."""
    inset = False                                    # read by PrismStyle: no margin round an inset ring

    def paintEvent(self, _):
        target = self.widget()
        if target is None:
            return
        p = QPainter(self)
        p.setRenderHint(QPainter.Antialiasing, True)
        ring = SIZE["focus_ring"]
        pen = QPen(TH.q("focus"), ring)
        pen.setJoinStyle(Qt.MiterJoin)
        p.setPen(pen); p.setBrush(Qt.NoBrush)
        r = QRectF(self.rect())
        if self.inset:
            left = int(target.property("ring_left") or 0)
            p.drawRect(r.adjusted(left + ring / 2, ring / 2, -ring / 2, -ring / 2))
        else:
            corner = SIZE["radius"] + SIZE["focus_gap"] + ring / 2
            p.drawRoundedRect(r.adjusted(ring / 2, ring / 2, -ring / 2, -ring / 2), corner, corner)


class FocusFollower(QObject):
    """Decides when the ring shows, from the reason the focus arrived: always after Tab, Shift+Tab, an arrow
    key within a group or a shortcut; after a mouse click only in a field one types in. A window coming back
    to the front keeps what it had. A widget with set_focus_shown() draws its own sign (a table rings its
    current row). One filter on the application; it looks at focus events only."""
    KEYS = (Qt.TabFocusReason, Qt.BacktabFocusReason, Qt.ShortcutFocusReason)
    TYPED_IN = (QLineEdit, QAbstractSpinBox)

    def __init__(self, app):
        super().__init__(app)
        self._ring, self._target, self._keyboard = None, None, False
        app.installEventFilter(self)

    def eventFilter(self, obj, ev):
        kind = ev.type()
        if kind == QEvent.FocusIn and obj.isWidgetType():
            if obj.window().windowType() in (Qt.Popup, Qt.ToolTip):      # an opened list or menu marks its own row
                return False
            reason = ev.reason()
            if reason in self.KEYS:
                self._keyboard = True
            elif reason == Qt.MouseFocusReason:
                self._keyboard = False
            self.show_on(obj if (self._keyboard or isinstance(obj, self.TYPED_IN)) else None)
        elif kind in (QEvent.FocusOut, QEvent.Hide) and obj is self._target:
            self.show_on(None)                       # also when its page or its dialog goes: before it can be deleted
        return False

    def target(self):
        if self._target is not None and not shiboken6.isValid(self._target):
            self._target = None
        return self._target

    def show_on(self, widget):
        old, self._target = self.target(), widget
        if old is not None and old is not widget and hasattr(old, "set_focus_shown"):
            old.set_focus_shown(False)
        ring = self._ring if (self._ring is not None and shiboken6.isValid(self._ring)) else None
        own = widget is not None and hasattr(widget, "set_focus_shown")
        if own:
            widget.set_focus_shown(True)
        if widget is None or own:
            if ring is not None:
                ring.setWidget(None)
            return
        if ring is None:                             # the frame moves itself into whichever window holds its control;
            ring = self._ring = FocusRing(widget.window())     # if that window was deleted, so was the frame: a new one
        ring.setWidget(None)
        ring.inset = widget.property("ring") == "inset"
        ring.setWidget(widget)
        ring.update()

    def refresh(self):
        if self.target() is not None:
            self.show_on(self._target)


# ── tables: three rules; two-line cells, a marker with its word and the selected row's bar are ONE delegate ──
def _wrapped(text, font, width, line_h):
    layout = QTextLayout(text, font)
    opt = QTextOption()
    opt.setWrapMode(QTextOption.WordWrap)
    layout.setTextOption(opt)
    layout.beginLayout()
    n = 0
    while True:
        ln = layout.createLine()
        if not ln.isValid():
            break
        ln.setLineWidth(width)
        ln.setPosition(QPointF(0, n * line_h + (line_h - ln.height()) / 2))
        n += 1
    layout.endLayout()
    return layout, n


class RuleDelegate(QStyledItemDelegate):
    """A cell is described by a small dict in Qt.UserRole:
         {"t": text, "quiet": bool, "elide": "middle" | "end", "wrap": bool}
         {"value": bold figure, "suffix": plain text after it, "sub": a small second line}   (right-aligned)
         {"state": (marker, word)}"""

    def _pads(self, index):
        last = index.column() == index.model().columnCount() - 1
        return (SIZE["table_lead"] if index.column() == 0 else 0), (0 if last else SIZE["table_col_gap"])

    def sizeHint(self, option, index):
        spec = index.data(Qt.UserRole) or {}
        view = self.parent()
        left, right = self._pads(index)
        width = view.column_px(index.column()) - left - right
        line_h = TYPE["body"][1]
        lines_h = line_h
        if spec.get("wrap"):
            lines_h = line_h * _wrapped(spec["t"], inter("body"), width, line_h)[1]
        elif spec.get("sub"):
            lines_h = line_h + TYPE["small"][1]
        return QSize(view.column_px(index.column()), lines_h + 2 * SIZE["table_pad"])

    def paint(self, p, option, index):
        spec = index.data(Qt.UserRole) or {}
        r = QRectF(option.rect)
        selected = bool(option.state & QStyle.State_Selected)
        hair, line_h, m = SIZE["hair"], TYPE["body"][1], SIZE["marker"]
        p.save()
        if selected:                                       # the tint, a line above and below, and the bar
            accent = TH.q("accent")
            p.fillRect(r, TH.q("accent_tint"))
            fill_px(p, QRectF(r.left(), r.top(), r.width(), hair), accent)
            fill_px(p, QRectF(r.left(), r.bottom() - hair, r.width(), hair), accent, from_bottom=True)
            if index.column() == 0:
                fill_px(p, QRectF(r.left(), r.top(), SIZE["bar"], r.height()), accent)
        left, right = self._pads(index)
        box = r.adjusted(left, SIZE["table_pad"], -right, -SIZE["table_pad"])
        line = QRectF(box.left(), box.top(), box.width(), line_h)
        if "state" in spec:
            kind, word = spec["state"]
            paint_marker(p, kind, QRectF(box.left(), box.top() + (line_h - m) / 2, m, m))
            p.setFont(inter("body")); p.setPen(TH.q("ink"))
            p.drawText(line.adjusted(m + SIZE["state_gap"], 0, 0, 0), Qt.AlignLeft | Qt.AlignVCenter, word)
        elif "value" in spec:
            plain, bold = inter("body"), inter("strong")
            suffix = spec.get("suffix", "")
            sw = QFontMetricsF(plain).horizontalAdvance(suffix)
            p.setPen(TH.q("ink"))
            p.setFont(plain); p.drawText(line, Qt.AlignRight | Qt.AlignVCenter, suffix)
            p.setFont(bold); p.drawText(line.adjusted(0, 0, -sw, 0), Qt.AlignRight | Qt.AlignVCenter, spec["value"])
            if spec.get("sub"):
                p.setFont(inter("small")); p.setPen(TH.q("ink_2"))
                p.drawText(QRectF(box.left(), box.top() + line_h, box.width(), TYPE["small"][1]),
                           Qt.AlignRight | Qt.AlignVCenter, spec["sub"])
        else:
            f = inter("strong" if (selected and index.column() == 0) else "body")
            p.setFont(f); p.setPen(TH.q("ink_2" if spec.get("quiet") else "ink"))
            text = spec.get("t", "")
            if spec.get("wrap"):
                layout, _ = _wrapped(text, f, box.width(), line_h)
                layout.draw(p, box.topLeft())
            else:
                mode = Qt.ElideMiddle if spec.get("elide") == "middle" else Qt.ElideRight
                p.drawText(line, Qt.AlignLeft | Qt.AlignVCenter, QFontMetricsF(f).elidedText(text, mode, box.width()))
        p.restore()


class RuleHeader(QHeaderView):
    """The header of a ruled table. It paints its own words, in the small semibold face, and the 1 px rule under
    them in one stroke across all columns: neither goes through the style sheet or the model."""

    def __init__(self, parent, aligns):
        super().__init__(Qt.Horizontal, parent)
        self._aligns = aligns
        self.setFixedHeight(TYPE["small"][1] + 2 * SIZE["table_pad"] + SIZE["hair"])
        self.setHighlightSections(False)
        self.setSectionsClickable(False)
        self.setMinimumSectionSize(SIZE["table_section_least"])

    def paintSection(self, p, rect, i):
        if not rect.isValid():
            return
        left = SIZE["table_lead"] if i == 0 else 0
        right = 0 if i == self.count() - 1 else SIZE["table_col_gap"]
        pad = SIZE["table_pad"]
        p.save()
        p.setFont(inter("small", 600))
        p.setPen(TH.q("ink"))
        p.drawText(QRectF(rect).adjusted(left, pad, -right, -(pad + SIZE["hair"])), self._aligns[i] | Qt.AlignVCenter,
                   str(self.model().headerData(i, Qt.Horizontal)))
        p.restore()

    def paintEvent(self, e):
        super().paintEvent(e)
        hair = SIZE["hair"]
        fill_px(QPainter(self.viewport()), QRectF(0, self.height() - hair, self.width(), hair), TH.q("rule"), from_bottom=True)


class RuleView(QTableView):
    """The rows of a ruled table. columns: (title, width | "fit" | None to stretch, alignment); rows: lists of
    cell dicts (see RuleDelegate). With the keyboard focus it rings its current row, not the whole table."""

    def __init__(self, columns, rows, selectable=True, selected=None):
        super().__init__()
        self._focus_shown = False
        self._widths = [self._fit(c, columns, rows) if w == "fit" else w for c, (_, w, _) in enumerate(columns)]
        m = QStandardItemModel(len(rows), len(columns), self)
        for c, (title, _, _) in enumerate(columns):
            m.setHeaderData(c, Qt.Horizontal, title)
        for r, row in enumerate(rows):
            for c, spec in enumerate(row):
                words = spec.get("t") or (spec.get("value", "") + spec.get("suffix", "")) or spec["state"][1]
                item = QStandardItem(words)
                item.setData(spec, Qt.UserRole)
                item.setEditable(False)
                m.setItem(r, c, item)
        self.setHorizontalHeader(RuleHeader(self, [a for _, _, a in columns]))
        self.setModel(m)
        self.setItemDelegate(RuleDelegate(self))
        self.setFrameShape(QFrame.NoFrame)
        self.setShowGrid(False)
        self.verticalHeader().hide()
        self.verticalHeader().setMinimumSectionSize(SIZE["table_row"])
        self.verticalHeader().setSectionResizeMode(QHeaderView.Fixed)
        hh = self.horizontalHeader()
        for c, width in enumerate(self._widths):
            hh.setSectionResizeMode(c, QHeaderView.Stretch if width is None else QHeaderView.Fixed)
            if width is not None:
                hh.resizeSection(c, width)
        self.setMinimumWidth(sum(w for w in self._widths if w is not None) + SIZE["table_stretch_least"])
        self.setSelectionBehavior(QAbstractItemView.SelectRows)
        self.setSelectionMode(QAbstractItemView.SingleSelection if selectable else QAbstractItemView.NoSelection)
        self.setFocusPolicy(Qt.StrongFocus if selectable else Qt.NoFocus)
        self.setEditTriggers(QAbstractItemView.NoEditTriggers)
        self.setHorizontalScrollBarPolicy(Qt.ScrollBarAlwaysOff)
        self.setVerticalScrollBarPolicy(Qt.ScrollBarAlwaysOff)
        self.setWordWrap(False)
        if selected is not None:
            self.selectRow(selected)

    @staticmethod
    def _fit(c, columns, rows):
        """A column as wide as its widest cell, measured in the toolkit's own letter widths (the first column
        in semibold, which is how its selected row is set), plus the space between columns."""
        body = QFontMetricsF(inter("strong" if c == 0 else "body"))
        need = QFontMetricsF(inter("small", 600)).horizontalAdvance(columns[c][0])
        for row in rows:
            spec = row[c]
            need = max(need, SIZE["marker"] + SIZE["state_gap"] + body.horizontalAdvance(spec["state"][1])
                       if "state" in spec else body.horizontalAdvance(spec.get("t", "")))
        lead = SIZE["table_lead"] if c == 0 else 0
        gap = 0 if c == len(columns) - 1 else SIZE["table_col_gap"]
        return min(SIZE["table_fit_most"], math.ceil(need)) + lead + gap

    def column_px(self, c):
        """A column's width as it WILL be: the stretching one is not yet resized when the rows are measured."""
        if self._widths[c] is not None:
            return self._widths[c]
        return max(SIZE["table_stretch_least"], self.width() - sum(w for w in self._widths if w is not None))

    def set_focus_shown(self, on):
        self._focus_shown = on
        self.viewport().update()

    def resizeEvent(self, e):
        super().resizeEvent(e)
        m, d = self.model(), self.itemDelegate()
        total = self.horizontalHeader().height()
        for r in range(m.rowCount()):                       # resizeRowsToContents() made every row 1 px too tall
            h = max(d.sizeHint(None, m.index(r, c)).height() for c in range(m.columnCount()))
            self.setRowHeight(r, h)
            total += h
        if self.height() != total:
            self.setFixedHeight(total)

    def paintEvent(self, e):
        super().paintEvent(e)
        index = self.currentIndex()
        if self._focus_shown and index.isValid():           # the ring round the current row, clear of its bar
            p = QPainter(self.viewport())
            p.setRenderHint(QPainter.Antialiasing, True)
            ring, inset = SIZE["focus_ring"], SIZE["row_ring_inset"] + SIZE["focus_ring"] / 2
            r = QRectF(0, self.rowViewportPosition(index.row()), self.viewport().width(), self.rowHeight(index.row()))
            p.setPen(QPen(TH.q("focus"), ring)); p.setBrush(Qt.NoBrush)
            p.drawRoundedRect(r.adjusted(SIZE["bar"] + inset, inset, -inset, -inset), SIZE["radius"], SIZE["radius"])

def journal_axes(fig, spec):
    """One pair of axes, placed by its four margins in pixels: it grows and shrinks with the figure."""
    w, h = fig.get_size_inches() * 100.0
    ax = fig.add_axes([spec["left"] / w, spec["bottom"] / h, 1 - (spec["left"] + spec["right"]) / w,
                       1 - (spec["top"] + spec["bottom"]) / h])
    ax.tick_params(pad=spec["tick_pad"] * PT)
    return ax


def name_axes(ax, x_name, y_name, spec):
    """The names of the axes as on a journal page: x centred under its axis; y upright at the head of its axis,
    beside it when the margin has room for it and above it, from the figure's left edge, when it has not."""
    F = FIGURE
    ax.annotate(x_name, (0.5, 0), xycoords="axes fraction", xytext=(0, -spec["x_name_drop"] * PT),
                textcoords="offset points", ha="center", va="baseline")
    if spec.get("y_name_above"):
        ax.annotate(y_name, (0, 1), xycoords=("figure fraction", "axes fraction"), xytext=(0, F["y_name_lift"] * PT),
                    textcoords="offset points", ha="left", va="baseline")
    else:
        ax.annotate(y_name, (0, 1), xycoords="axes fraction", xytext=(-F["y_name_gap"] * PT, -F["y_name_sink"] * PT),
                    textcoords="offset points", ha="right", va="baseline")


def label_line(ax, x, y, text, dx=None, dy=0.0, ha="left", va="center"):

    dx = FIGURE["line_label_gap"] if dx is None else dx
    return ax.annotate(text, (x, y), xytext=(dx * PT, dy * PT), textcoords="offset points", ha=ha, va=va,

class Plot(QWidget):
    """A matplotlib figure rendered by Agg at the width its layout gives it and at the scaling of the screen it
    is on. It is rendered when it is painted and kept until its size, its screen's scaling, the theme or its
    options change."""

    def __init__(self, spec, draw, name):
        super().__init__()
        self._draw, self._kw, self._key, self._pm = draw, {}, None, None
        self.setFixedHeight(spec["h"])
        self.setMinimumWidth(spec["min_w"])
        self.setSizePolicy(QSizePolicy.Expanding, QSizePolicy.Fixed)
        self.setAccessibleName(name)

    def set_options(self, **kw):
        self._kw = kw
        self.update()

    def retheme(self):
        self._key = None
        self.update()

    def event(self, ev):
        if ev.type() == QEvent.DevicePixelRatioChange:     # moved to a screen with another scaling
            self.update()
        return super().event(ev)

    def paintEvent(self, _):
        dpr = self.devicePixelRatioF()
        key = (self.width(), self.height(), dpr, TH.dark, tuple(sorted(self._kw.items())))
        if key != self._key:
            fig = Figure(figsize=(self.width() / 100, self.height() / 100), dpi=100 * dpr)
            canvas = FigureCanvasAgg(fig)
            self._draw(fig, **self._kw)
            canvas.draw()
            buf = canvas.buffer_rgba()
            img = QImage(buf, buf.shape[1], buf.shape[0], buf.shape[1] * 4, QImage.Format_RGBA8888).copy()
            img.setDevicePixelRatio(dpr)
            self._pm, self._key = QPixmap.fromImage(img), key
        QPainter(self).drawPixmap(0, 0, self._pm)

class IntField(QLineEdit):
    """A whole-number field of the kind PRISM has: a line edit, not a spin box (no stepping on the wheel or the
    arrow keys). Right-aligned; four digits or more carry the separator, also while they are typed."""

    def __init__(self, value=0):
        super().__init__(f"{value:,}")
        self.setAlignment(Qt.AlignRight | Qt.AlignVCenter)
        self.textEdited.connect(self._regroup)

    def _regroup(self, text):
        before = sum(ch.isdigit() for ch in text[:self.cursorPosition()])      # digits left of the caret
        digits = "".join(ch for ch in text if ch.isdigit())
        shown = f"{int(digits):,}" if digits else ""
        self.setText(shown)
        pos = 0
        while pos < len(shown) and before > 0:
            before -= shown[pos].isdigit()
            pos += 1
        self.setCursorPosition(pos)

    def value_or_none(self):
        digits = self.text().replace(",", "").strip()
        return int(digits) if digits.isdigit() else None

class Scroller(QScrollArea):
    """The middle of a page. It scrolls when the window is too small for it, and tells what it holds how wide
    the page is so that two columns can become one. It keeps the ring's room round what it holds, so a ring at
    the edge is whole; it is a stop for the Tab key only while there is something to scroll."""

    def __init__(self, body):
        super().__init__()
        self.body = body
        self.setWidgetResizable(True)
        self.setFrameShape(QFrame.NoFrame)
        self.setHorizontalScrollBarPolicy(Qt.ScrollBarAlwaysOff)
        self.setFocusPolicy(Qt.NoFocus)
        self.setProperty("ring", "inset")
        holder = QWidget()
        self._v = QVBoxLayout(holder)
        self._v.setSpacing(0)
        self._v.addWidget(body)
        self._v.addStretch(1)
        self._margins(False)
        self.setWidget(holder)
        self.verticalScrollBar().rangeChanged.connect(lambda lo, hi: self._margins(hi > lo))

    def _margins(self, scrolls):
        room = ring_room()
        self._v.setContentsMargins(room, 0, room + (SIZE["scroll_gap"] if scrolls else 0), room)
        self.setFocusPolicy(Qt.TabFocus if scrolls else Qt.NoFocus)

    def resizeEvent(self, e):
        self.body.fit(self.width() - 2 * ring_room() - SIZE["scroll_bar"] - SIZE["scroll_gap"])
        super().resizeEvent(e)


class Columns(QWidget):
    """Two blocks side by side while the page is wide enough for both, one above the other when it is not."""

    def __init__(self, left, right, stretch, gap, gap_stacked):
        super().__init__()
        self._parts, self._gaps = (left, right), (gap, gap_stacked)
        self.box = QBoxLayout(QBoxLayout.LeftToRight, self)
        self.box.setContentsMargins(0, 0, 0, 0); self.box.setSpacing(gap)
        self.box.addWidget(left, stretch[0])
        self.box.addWidget(right, stretch[1])

    def fit(self, width):
        need = sum(w.minimumSizeHint().width() for w in self._parts) + self._gaps[0]
        wide = width >= max(SIZE["two_columns_from"], need)
        want = QBoxLayout.LeftToRight if wide else QBoxLayout.TopToBottom
        if self.box.direction() != want:
            self.box.setDirection(want)
            self.box.setSpacing(self._gaps[0 if wide else 1])


class Stages(QWidget):
    """The stages of a run, each a marker, a name and a state: one row, or two rows of two when a row is too long."""

    def __init__(self, stages):
        super().__init__()
        self.items = []
        for kind, name, words in stages:
            one = QWidget()
            h = QHBoxLayout(one)
            h.setContentsMargins(0, 0, 0, 0); h.setSpacing(GAP["stage_parts"])
            h.addWidget(Marker(kind))
            h.addWidget(label(name, "strong"))
            h.addWidget(label(words, None if kind == "now" else "quiet"))
            self.items.append(one)
        self.grid = QGridLayout(self)
        self.grid.setContentsMargins(0, 0, 0, 0)
        self.grid.setHorizontalSpacing(GAP["stages"][0]); self.grid.setVerticalSpacing(GAP["stages"][1])
        self.setSizePolicy(QSizePolicy.Ignored, QSizePolicy.Fixed)      # its width is the page's, not its own
        self._cols = 0
        self._lay(len(self.items))

    def _lay(self, cols):
        if cols == self._cols:
            return
        for w in self.items:
            self.grid.removeWidget(w)
        for i, w in enumerate(self.items):
            self.grid.addWidget(w, i // cols, i % cols, Qt.AlignLeft)
        for c in range(len(self.items) + 1):
            self.grid.setColumnStretch(c, 1 if c == cols else 0)
        self._cols = cols

    def resizeEvent(self, e):
        super().resizeEvent(e)
        need = sum(w.sizeHint().width() for w in self.items) + (len(self.items) - 1) * GAP["stages"][0]
        self._lay(len(self.items) if need <= self.width() else 2)

class RunBody(QWidget):
    """The run page under its status. Wide: the figure and the records beside the health, the messages and one
    recording. Narrow: one column in reading order (figure, health, messages, recording, records). Only the
    records move, between the left column and the foot; nothing changes its parent."""

    def __init__(self, figure, records, side):
        super().__init__()
        self._parts, self._wide = (figure, records, side), True
        left, right, gap = GAP["run_columns"]
        self.outer = QBoxLayout(QBoxLayout.LeftToRight, self)
        self.outer.setContentsMargins(0, 0, 0, 0); self.outer.setSpacing(gap)
        self.left = QVBoxLayout()
        self.left.setSpacing(GAP["block"])
        self.left.addWidget(figure); self.left.addWidget(records); self.left.addStretch(1)
        self.outer.addLayout(self.left, left)
        self.outer.addWidget(side, right)

    def fit(self, width):
        figure, records, side = self._parts
        gap = GAP["run_columns"][2]
        need = max(figure.minimumSizeHint().width(), records.minimumSizeHint().width()) + gap + side.minimumSizeHint().width()
        wide = width >= max(SIZE["two_columns_from"], need)
        if wide == self._wide:
            return
        self._wide = wide
        if wide:
            self.outer.removeWidget(records); self.left.insertWidget(1, records)
        else:
            self.left.removeWidget(records); self.outer.addWidget(records)
        self.outer.setDirection(QBoxLayout.LeftToRight if wide else QBoxLayout.TopToBottom)
        self.outer.setSpacing(gap if wide else GAP["block"])

class Ask(QDialog):
    """The one way the program asks or tells: the toolkit's own dialog window in the system's frame, a marker,
    one sentence in the book face, and stock buttons. The safe answer is the default and STAYS the default:
    Enter, Escape and the frame's close button all mean it, wherever the keyboard focus is (the answer that
    loses work never takes the default when the focus reaches it; it is pressed with the mouse or the space
    bar). The accent comes from the button's class, so it does not move with the focus either."""

    def __init__(self, parent, title, sentence, safe, risky=None, marker="warn"):
        super().__init__(parent)
        set_class(self, cls="ask")
        self.setWindowTitle(title)
        self.setWindowModality(Qt.WindowModal)
        self.setAttribute(Qt.WA_DeleteOnClose, True)
        left, top, right, bottom = DIALOG["pad"]
        v = QVBoxLayout(self)
        v.setContentsMargins(left, top, right, bottom); v.setSpacing(DIALOG["gap"])
        v.setSizeConstraint(QLayout.SetFixedSize)
        body = QHBoxLayout()
        body.setSpacing(DIALOG["marker_gap"])
        mv = QVBoxLayout()
        mv.setContentsMargins(0, DIALOG["marker_drop"], 0, 0)
        mv.addWidget(Marker(marker, DIALOG["marker"])); mv.addStretch(1)
        body.addLayout(mv)
        self.sentence = Rich(sentence, lines="book", wrap=True)
        self.sentence.setFixedWidth(DIALOG["w"] - left - right - DIALOG["marker"] - DIALOG["marker_gap"])
        body.addWidget(self.sentence)
        v.addLayout(body)
        answers = QHBoxLayout()                                     # the two answers at opposite ends
        answers.setSpacing(GAP["action"])
        self.safe = button(safe, "primary")
        self.safe.setDefault(True)
        self.safe.clicked.connect(self.reject)
        self.risky = None
        if risky:
            self.risky = button(risky, "danger", "stop")
            self.risky.setAutoDefault(False)
            self.risky.clicked.connect(self.accept)
            answers.addWidget(self.risky)
        answers.addStretch(1)
        answers.addWidget(self.safe)
        v.addLayout(answers)
        self.safe.setFocus(Qt.TabFocusReason)


def ask(parent, title, sentence, safe, risky=None, marker="warn"):
    """Build the question; the caller connects accepted (the risky answer) and calls open(). Never exec()."""
    return Ask(parent, title, sentence, safe, risky, marker)


def ask_stop(parent):
    return ask(parent, "Stop training?", "The simulations are kept. The fit so far, 62 epochs and 65 hours, "
                                         "will be lost and cannot be resumed.",
               safe="Keep training", risky="Stop and lose the fit")

def set_class_fonts():
    """Setting an application style sheet puts the SYSTEM's font (Segoe UI) back on lists and tables, their
    headers, message boxes, menus, tool tips and status bars, whatever the application font is. So each is
    told again after every setStyleSheet (that is, after every change of theme)."""
    for cls in ("QAbstractItemView", "QHeaderView", "QMessageBox", "QMenu", "QMenuBar", "QTipLabel", "QStatusBar"):
        QApplication.setFont(inter("body"), cls)


def pump(app, n=6):
    for _ in range(n):
        app.sendPostedEvents()
        app.processEvents()


def hidden(widget):
    """Show a top-level widget without anything reaching the owner's screen."""
    widget.setAttribute(Qt.WA_DontShowOnScreen, True)
    widget.show()
    return widget


def press_tab(app, old, new):
    """What the Tab key does to two controls, without a keyboard: the focus leaves one and arrives at the other."""
    QApplication.sendEvent(old, QFocusEvent(QEvent.FocusOut, Qt.TabFocusReason))
    QApplication.sendEvent(new, QFocusEvent(QEvent.FocusIn, Qt.TabFocusReason))
    pump(app)


def where(widget, top):
    return QRect(widget.mapTo(top, QPoint(0, 0)), widget.size())


def focus_sheet(app, win, path):
    """The ring on each kind of control, cut from the window and set side by side on one sheet."""
    S, R = win.setup, win.run
    shots = [("setup", "A number field", S.batches), ("setup", "A field in error", S.name), ("setup", "A drop-down", S.choice),
             ("setup", "A tick box", S.keep), ("setup", "A folded line", S.fold), ("setup", "A button", S.gate.check),
             ("setup", "A button with an icon", S.help), ("run", "An action that loses work", R.stop),
             ("run", "A radio button", R.recent), ("run", "A link", R.link),
             ("setup", "A part of the program", win.rail.parts[4]), ("setup", "A step", win.rail.steps[3]),
             ("setup", "The step in hand", win.rail.steps[2]), ("run", "A table: its current row", R.records.view)]
    crops = []
    for page, words, w in shots:
        win.set_page(page)
        pump(app)
        FOLLOWER.show_on(w)
        pump(app)
        crops.append((words, win.grab(where(w, win).adjusted(-10, -8, 10, 8).intersected(win.rect()))))
    FOLLOWER.show_on(None)
    dpr = crops[0][1].devicePixelRatio()
    pad, cap_h, sheet_w = 16, 18, 960
    places, x, y, row_h = [], pad, pad, 0
    for _, pm in crops:
        w, h = pm.width() / dpr, pm.height() / dpr
        if x + w + pad > sheet_w and x > pad:
            x, y, row_h = pad, y + row_h + pad, 0
        places.append((x, y))
        x, row_h = x + w + pad, max(row_h, h + cap_h)
    sheet = QPixmap(int(sheet_w * dpr), int((y + row_h + pad) * dpr))
    sheet.setDevicePixelRatio(dpr)
    sheet.fill(TH.q("paper"))
    p = QPainter(sheet)
    p.setFont(inter("small")); p.setPen(TH.q("ink_2"))
    for (words, pm), (x, y) in zip(crops, places):
        p.drawText(QRectF(x, y, 400, cap_h - 2), Qt.AlignLeft | Qt.AlignVCenter, words)
        p.drawPixmap(QPointF(x, y + cap_h), pm)
    p.end()
    sheet.save(path)
